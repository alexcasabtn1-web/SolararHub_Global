    -- Função ultra segura: deleta procurando diretamente pelo nome
local function deletarBloqueios()
    local deletouPart = false
    local deletouBlocker = false

    -- Varre absolutamente tudo dentro da pasta Workspace.Build
    local buildFolder = workspace:FindFirstChild("Build") or workspace:FindFirstChild("build")
    
    if buildFolder then
        for _, objeto in pairs(buildFolder:GetDescendants()) do
            -- Verifica se encontrou a 'Part' que fica dentro de 'BLOCKER'
            if objeto.Name == "Part" and objeto.Parent and objeto.Parent.Name == "BLOCKER" then
                objeto:Destroy()
                deletouPart = true
            end
            
            -- Verifica se encontrou o 'Blocker' que fica dentro de 'ConvenienceS'
            if objeto.Name == "Blocker" and objeto.Parent and objeto.Parent.Name == "ConvenienceS" then
                objeto:Destroy()
                deletouBlocker = true
            end
        end
    end

    -- Mostra o resultado real na tela para você saber se deu certo
    if deletouPart or deletouBlocker then
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "Limpeza Concluída",
            Text = "Os bloqueios foram removidos com sucesso!",
            Duration = 3
        })
    else
        -- Aviso discreto caso o jogo já tenha deletado ou mudado o nome das pastas
        print("Avisos de caminho ignorados. Objetos já limpos ou não gerados.")
    end
end

-- Executa a limpeza direto
deletarBloqueios()
    end
})








-- Botão para clonar qualquer objeto com o formato original idêntico
-- Novo Transmutador que preserva o som de comer
DivertidasTab:AddButton({
    Name = "Objeto virar comida(Usa o 1º Item do Inventário)",
    Callback = function()
        local player = game:GetService("Players").LocalPlayer
        local mouse = player:GetMouse()
        local character = player.Character
        local backpack = player:FindFirstChild("Backpack")
        
        -- 1. Encontra de forma inteligente qualquer ferramenta equipada na mão primeiro
        local tool = character and character:FindFirstChildWhichIsA("Tool")
        
        -- 2. Se não estiver na mão, pega a primeiríssima ferramenta que estiver na Backpack
        if not tool and backpack then
            tool = backpack:FindFirstChildWhichIsA("Tool")
        end
        
        -- Se o jogador estiver com o inventário 100% vazio
        if not tool then
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "Erro",
                Text = "Pegue ou compre qualquer comida primeiro!",
                Duration = 3
            })
            return
        end
        
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "Transmutar Universal",
            Text = "Usando: " .. tool.Name .. ".\nClique em um objeto do mapa!",
            Duration = 4
        })
        
        local conexao
        conexao = mouse.Button1Down:Connect(function()
            local alvo = mouse.Target
            
            if alvo and alvo:IsA("BasePart") and alvo.Name ~= "Terrain" then
                conexao:Disconnect()
                
                local antigoHandle = tool:FindFirstChild("Handle")
                local somComer = nil
                
                -- Se o Handle antigo existir, salva o som dele (independente do nome) antes de deletar
                if antigoHandle then
                    somComer = antigoHandle:FindFirstChildWhichIsA("Sound")
                    if somComer then
                        somComer.Parent = tool
                    end
                    antigoHandle:Destroy()
                end
                
                if not somComer then
                    somComer = tool:FindFirstChildWhichIsA("Sound")
                end
                
                -- Clona o formato do objeto clicado
                local novoHandle = alvo:Clone()
                novoHandle.Name = "Handle"
                novoHandle.CanCollide = false
                novoHandle.Anchored = false 
                novoHandle.CFrame = CFrame.new(0, 0, 0)
                novoHandle.Parent = tool
                
                -- Devolve o som de comer original daquela comida específica para o novo Handle
                if somComer then
                    somComer.Parent = novoHandle
                end
                
                game:GetService("StarterGui"):SetCore("SendNotification", {
                    Title = "Sucesso! 🍽️",
                    Text = "Objeto injetado em: " .. tool.Name .. "! Reequipe o item.",
                    Duration = 3
                })
            end
        end)
    end
})

-- DISPARADOR DO BOTÃO START COM O SEU TRECHO DE CÓDIGO FECHADO CORRETAMENTE
BtnStart.MouseButton1Click:Connect(function()
    local Players = game:GetService("Players")
    local Workspace = game:GetService("Workspace")
    local localPlayer = Players.LocalPlayer
    local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local rootPart = character:WaitForChild("HumanoidRootPart")

    for _, v in ipairs(Workspace:GetChildren()) do
        if v.Name == "Gato_Exercito_GUI" then v:Destroy() end
    end

    local pastaAlvo = Workspace:FindFirstChild("build") and Workspace.build:FindFirstChild("private rooms")
    if not pastaAlvo then return end

    local originalModel = pastaAlvo:FindFirstChild(gatoSelecionado)
    if not originalModel then
    warn("O gato '" .. gatoSelecionado .. "' nao foi encontrado na pasta private rooms!")
    return
end

for i = 0, (quantidadeGatos - 1) do
    local cloneGato = originalModel:Clone()
    cloneGato.Name = "Gato_Exercito_GUI"
    
    local gatoRoot = cloneGato:IsA("Model") and (cloneGato.PrimaryPart or cloneGato:FindFirstChildWhichIsA("BasePart")) or cloneGato
    if not gatoRoot then continue end

    local deslocamentoX, deslocamentoZ
    
    if modoSelecionado == "Vertical" then
        local posicaoNaFila = i % 3
        local numeroDaFilaLateral = math.floor(i / 3)
        deslocamentoX = numeroDaFilaLateral * 2.5
        deslocamentoZ = 3 + (posicaoNaFila * 3)
    else
        local posicaoHorizontal = i % 3
        local fileiraTraseira = math.floor(i / 3)
        deslocamentoX = (posicaoHorizontal - 1) * 2.5
        deslocamentoZ = 3 + (fileiraTraseira * 3)
    end

    gatoRoot.CFrame = rootPart.CFrame * CFrame.new(deslocamentoX, 0, deslocamentoZ)
    cloneGato.Parent = Workspace

    for _, part in ipairs(cloneGato:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end

    local attachmentGato = Instance.new("Attachment", gatoRoot)
    local attachmentPlayer = Instance.new("Attachment", rootPart)
    attachmentPlayer.Position = Vector3.new(deslocamentoX, -1, deslocamentoZ)

    local alinhamento = Instance.new("AlignPosition")
    alinhamento.MaxForce = 50000
    alinhamento.Responsiveness = 15
    alinhamento.Attachment0 = attachmentGato
    alinhamento.Attachment1 = attachmentPlayer
    alinhamento.Parent = gatoRoot

    local orientacao = Instance.new("AlignOrientation")
    orientacao.MaxTorque = 50000
    orientacao.Responsiveness = 15
    orientacao.Attachment0 = attachmentGato
    orientacao.Attachment1 = attachmentPlayer
    orientacao.Parent = gatoRoot
end

print("Exercito iniciado com " .. quantidadeGatos .. "x " .. gatoSelecionado .. " em formato " .. modoSelecionado)
end)

        -- 1. Smartly finds any equipped tool in the player's hand first

        -- 2. If not in hand, gets the first tool found in the Backpack
        if not tool and backpack then
            tool = backpack:FindFirstChildWhichIsA("Tool")
        end
        
        -- If the player has a completely empty inventory
        if not tool then
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "Error",
                Text = "Get or buy any food first!",
                Duration = 3
            })
            return
        end
        
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "Universal Transmutation",
            Text = "Using: " .. tool.Name .. ".\nClick an object on the map!",
            Duration = 4
        })

                connection:Disconnect()
                
                local oldHandle = tool:FindFirstChild("Handle")
                local eatingSound = nil

-- START BUTTON TRIGGER WITH THE CODE SECTION PROPERLY CLOSED
BtnStart.MouseButton1Click:Connect(function()
    local Players = game:GetService("Players")
    local Workspace = game:GetService("Workspace")
    local localPlayer = Players.LocalPlayer
    local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local rootPart = character:WaitForChild("HumanoidRootPart")

    for _, v in ipairs(Workspace:GetChildren()) do
        if v.Name == "Cat_Army_GUI" then v:Destroy() end
    end

    local targetFolder = Workspace:FindFirstChild("build") and Workspace.build:FindFirstChild("private rooms")
    if not targetFolder then return end

    local originalModel = targetFolder:FindFirstChild(selectedCat)
    if not originalModel then
        warn("The cat '" .. selectedCat .. "' was not found in the private rooms folder!")
        return
    end

    for i = 0, (catQuantity - 1) do
        local clonedCat = originalModel:Clone()
        clonedCat.Name = "Cat_Army_GUI"
        
        local catRoot = clonedCat:IsA("Model") and (clonedCat.PrimaryPart or clonedCat:FindFirstChildWhichIsA("BasePart")) or clonedCat
        if not catRoot then continue end

        local offsetX, offsetZ
        
        if selectedMode == "Vertical" then
            local positionInLine = i % 3
            local sideRowNumber = math.floor(i / 3)
            offsetX = sideRowNumber * 2.5
            offsetZ = 3 + (positionInLine * 3)
        else
            local horizontalPosition = i % 3
            local backRow = math.floor(i / 3)
            offsetX = (horizontalPosition - 1) * 2.5
            offsetZ = 3 + (backRow * 3)
        end

        catRoot.CFrame = rootPart.CFrame * CFrame.new(offsetX, 0, offsetZ)
        clonedCat.Parent = Workspace

        for _, part in ipairs(clonedCat:GetDescendants()) do
            if part:IsA("BasePart") then 
                part.CanCollide = false 
            end
        end

        local catAttachment = Instance.new("Attachment", catRoot)
        local playerAttachment = Instance.new("Attachment", rootPart)
        playerAttachment.Position = Vector3.new(offsetX, -1, offsetZ)

        local alignment = Instance.new("AlignPosition")
        alignment.MaxForce = 50000
        alignment.Responsiveness = 15
        alignment.Attachment0 = catAttachment
        alignment.Attachment1 = playerAttachment
        alignment.Parent = catRoot

        local orientation = Instance.new("AlignOrientation")
        orientation.MaxTorque = 50000
        orientation.Responsiveness = 15
        orientation.Attachment0 = catAttachment
        orientation.Attachment1 = playerAttachment
        orientation.Parent = catRoot
    end

    print("Army started with " .. catQuantity .. "x " .. selectedCat .. " in " .. selectedMode .. " formation")
end)

end
})

                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)

            end

                createHighlight(player.Character)
            end

        if not _G.ESP_Connected then

            Players.PlayerAdded:Connect(function(player)
                player.CharacterAdded:Connect(function(char)
                    task.wait(1)
                    if _G.ESP_Active and player ~= LocalPlayer then

                    end
                end)
            end)

            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer then
                    player.CharacterAdded:Connect(function(char)
                        task.wait(1)

                    end)
                end
            end
        end
    end
